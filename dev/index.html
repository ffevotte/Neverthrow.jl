Neverthrow.jl

using Neverthrow

map_ok, map

plain function application : map is an alias for map_ok 

julia> square(x::Real) = x^2square (generic function with 1 method)
julia> map(square, Ok(2))Ok{Int64}(4)
julia> map(square, Err("oops"))Err{String}("oops")

chaining / piping calls : map can't be used like this, use map_ok instead

julia> Err("oops") |> map_ok(square)Err{String}("oops")

map_err

julia> Ok(:success) |> map_err(square)Ok{Symbol}(:success)
julia> Err(3) |> map_err(square)Err{Int64}(9)

unwrap_or

julia> Ok(2) |> map_ok(square) |> unwrap_or(10)4
julia> Err("oops") |> map_ok(square) |> unwrap_or(10)10

and_then

chaining results

julia> sq(x) = Ok(x^2)sq (generic function with 1 method)
julia> (Ok(2)
        |> and_then(sq)
        |> and_then(sq))Ok{Int64}(16)
julia> (Ok(2)
        |> and_then(sq)
        |> and_then(Err))Err{Int64}(4)
julia> (Ok(2)
        |> and_then(Err)
        |> and_then(sq))Err{Int64}(2)
julia> (Err("oops")
        |> and_then(sq)
        |> and_then(sq))Err{String}("oops")

flattening nested results

julia> Ok(Ok(42)) |> and_then(identity)Ok{Int64}(42)

or_else

           return Err(500)

julia> recover(Err(:not_found))Ok{String}("User does not exist")
julia> recover(Err(:pool_exhausted))Err{Int64}(500)
